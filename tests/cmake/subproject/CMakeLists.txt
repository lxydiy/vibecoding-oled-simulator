cmake_minimum_required(VERSION 3.20)
project(OledSubprojectCheck LANGUAGES C)
# Internal CMake integration fixture; not another product/repository.
set(CMAKE_C_STANDARD 99)
set(BUILD_TESTING ON)
add_subdirectory("${OLED_REPOSITORY}" oled)
if(TARGET oled_simulator OR TARGET test_framebuffer)
    message(FATAL_ERROR "Dependency unexpectedly builds examples or tests")
endif()
if(NOT CMAKE_C_STANDARD EQUAL 99)
    message(FATAL_ERROR "Dependency changed the parent's C standard")
endif()
if(NOT TARGET oled::sim)
    message(FATAL_ERROR "Public library target is missing")
endif()
if(WIN32)
    # Reuse an existing API test as the consumer; do not create a new app.
    add_executable(subproject_api_check "${OLED_REPOSITORY}/tests/test_win32.c")
    target_link_libraries(subproject_api_check PRIVATE oled::sim user32 gdi32)
endif()
