cmake_minimum_required(VERSION 3.20)
project(HeaterOLEDSimulator VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)

if(WIN32)
    set(PLATFORM_SOURCE src/platform_win32.c)
    set(PLATFORM_LIBRARIES user32 gdi32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(X11 REQUIRED)
    find_package(Threads REQUIRED)
    set(PLATFORM_SOURCE src/platform_x11.c)
    set(PLATFORM_LIBRARIES X11::X11 Threads::Threads)
else()
    message(FATAL_ERROR "Supported platforms: Windows and Linux (X11)")
endif()

add_library(oled_objects OBJECT src/oled_fb.c src/sim_backend.c ${PLATFORM_SOURCE})
set_target_properties(oled_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_include_directories(oled_objects PUBLIC include PRIVATE src)
target_link_libraries(oled_objects PUBLIC ${PLATFORM_LIBRARIES})

# The DLL/shared library exposes only the original ten OLED functions and
# the separately requested switch-state query. No backend symbols are exported.
add_library(oled_sim SHARED $<TARGET_OBJECTS:oled_objects>)
set_target_properties(oled_sim PROPERTIES EXPORT_NAME sim)
add_library(oled::sim ALIAS oled_sim)
target_include_directories(oled_sim PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_link_libraries(oled_sim PRIVATE ${PLATFORM_LIBRARIES})
if(WIN32)
    target_sources(oled_sim PRIVATE src/oled_sim.def)
endif()

# The executable host owns the event-loop lifecycle; application drawing code
# sees only public headers. Normal consumers can link oled::sim with their main.
add_executable(oled_simulator src/simulator_main.c examples/demo.c)
target_include_directories(oled_simulator PRIVATE src examples)
target_link_libraries(oled_simulator PRIVATE oled_objects)

function(strict_c target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
strict_c(oled_objects)
strict_c(oled_simulator)

if(BUILD_TESTING)
    add_executable(test_framebuffer tests/test_framebuffer.c src/oled_fb.c)
    target_include_directories(test_framebuffer PRIVATE include src)
    strict_c(test_framebuffer)
    add_test(NAME framebuffer COMMAND test_framebuffer)
    if(WIN32)
        add_executable(test_win32 tests/test_win32.c)
        target_link_libraries(test_win32 PRIVATE oled_sim user32 gdi32)
        strict_c(test_win32)
        add_test(NAME win32_window_and_keys COMMAND test_win32)
        add_test(NAME win32_host_exit COMMAND test_win32 --exit-open)
        add_test(NAME demo_smoke COMMAND oled_simulator --smoke)
        set_tests_properties(win32_window_and_keys win32_host_exit demo_smoke PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
    endif()
endif()

install(TARGETS oled_sim EXPORT oledTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT oledTargets NAMESPACE oled:: DESTINATION lib/cmake/oled_sim)
