cmake_minimum_required(VERSION 3.20)
project(HeaterOLEDSimulator VERSION 1.0.0 LANGUAGES C)

set(OLED_SIM_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(OLED_SIM_TOP_LEVEL ON)
    include(CTest)
endif()
option(OLED_SIM_BUILD_EXAMPLE "Build the OLED example application" ${OLED_SIM_TOP_LEVEL})
set(OLED_SIM_TESTS_DEFAULT OFF)
if(OLED_SIM_TOP_LEVEL AND BUILD_TESTING)
    set(OLED_SIM_TESTS_DEFAULT ON)
endif()
option(OLED_SIM_BUILD_TESTS "Build OLED simulator tests" ${OLED_SIM_TESTS_DEFAULT})
option(OLED_SIM_BUILD_SHARED "Build a shared library instead of a static library" OFF)

if(WIN32)
    set(PLATFORM_SOURCE src/platform_win32.c)
    set(PLATFORM_LIBRARIES user32 gdi32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(X11 REQUIRED)
    find_package(Threads REQUIRED)
    set(PLATFORM_SOURCE src/platform_x11.c)
    set(PLATFORM_LIBRARIES X11::X11 Threads::Threads)
else()
    message(FATAL_ERROR "Supported platforms: Windows and Linux (X11)")
endif()

add_library(oled_objects OBJECT src/oled_fb.c src/oled_log.c src/sim_backend.c ${PLATFORM_SOURCE})
set_target_properties(oled_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden C_EXTENSIONS OFF)
target_compile_features(oled_objects PUBLIC c_std_11)
target_include_directories(oled_objects PUBLIC include PRIVATE src)
target_link_libraries(oled_objects PUBLIC ${PLATFORM_LIBRARIES})

# The DLL/shared library exposes only the original ten OLED functions and
# switch-state query and log writer. No window/backend symbols are exported.
if(OLED_SIM_BUILD_SHARED)
    add_library(oled_sim SHARED $<TARGET_OBJECTS:oled_objects>)
else()
    add_library(oled_sim STATIC $<TARGET_OBJECTS:oled_objects>)
endif()
set_target_properties(oled_sim PROPERTIES EXPORT_NAME sim)
add_library(oled::sim ALIAS oled_sim)
target_include_directories(oled_sim PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_compile_features(oled_sim PUBLIC c_std_11)
target_link_libraries(oled_sim PRIVATE ${PLATFORM_LIBRARIES})
if(WIN32 AND OLED_SIM_BUILD_SHARED)
    target_sources(oled_sim PRIVATE src/oled_sim.def)
endif()

# The executable host owns the event-loop lifecycle; application drawing code
# sees only public headers. Normal consumers can link oled::sim with their main.
if(OLED_SIM_BUILD_EXAMPLE)
    add_executable(oled_simulator src/simulator_main.c examples/demo.c)
    target_include_directories(oled_simulator PRIVATE src examples)
    target_link_libraries(oled_simulator PRIVATE oled_objects)
endif()

function(strict_c target)
    target_compile_features(${target} PRIVATE c_std_11)
    set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()
strict_c(oled_objects)
if(OLED_SIM_BUILD_EXAMPLE)
    strict_c(oled_simulator)
endif()

if(OLED_SIM_BUILD_TESTS)
    enable_testing()
    add_executable(test_framebuffer tests/test_framebuffer.c src/oled_fb.c)
    target_include_directories(test_framebuffer PRIVATE include src)
    strict_c(test_framebuffer)
    add_test(NAME framebuffer COMMAND test_framebuffer)
    foreach(mode IN ITEMS full range)
        add_executable(test_log_${mode} tests/test_log.c tests/test_log_other.c)
        target_link_libraries(test_log_${mode} PRIVATE oled_sim)
        if(mode STREQUAL "full")
            target_compile_definitions(test_log_${mode} PRIVATE OLED_LOG_LEVEL=5)
        else()
            target_compile_definitions(test_log_${mode} PRIVATE OLED_LOG_MIN_LEVEL=2 OLED_LOG_LEVEL=4 OLED_LOG_LOCAL_LEVEL=3)
        endif()
        strict_c(test_log_${mode})
        add_test(NAME log_${mode} COMMAND ${CMAKE_COMMAND}
            "-DEXECUTABLE=$<TARGET_FILE:test_log_${mode}>" "-DMODE=${mode}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_log.cmake")
    endforeach()
    foreach(mode IN ITEMS none stub)
        add_executable(test_log_${mode} tests/test_log_disabled.c)
        if(mode STREQUAL "none")
            target_include_directories(test_log_${mode} PRIVATE include)
            target_compile_definitions(test_log_${mode} PRIVATE OLED_LOG_LEVEL=0)
        else()
            target_include_directories(test_log_${mode} PRIVATE examples/no_logging)
            # Even an explicitly enabled level must not enable the stub.
            target_compile_definitions(test_log_${mode} PRIVATE OLED_LOG_LEVEL=5)
        endif()
        strict_c(test_log_${mode})
        add_test(NAME log_${mode} COMMAND ${CMAKE_COMMAND}
            "-DEXECUTABLE=$<TARGET_FILE:test_log_${mode}>"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_no_log.cmake")
    endforeach()
    if(WIN32)
        add_executable(test_win32 tests/test_win32.c)
        target_link_libraries(test_win32 PRIVATE oled_sim user32 gdi32)
        strict_c(test_win32)
        add_test(NAME win32_window_and_keys COMMAND test_win32)
        add_test(NAME win32_host_exit COMMAND test_win32 --exit-open)
        set_tests_properties(win32_window_and_keys win32_host_exit PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
        if(OLED_SIM_BUILD_EXAMPLE)
            add_test(NAME demo_smoke COMMAND ${CMAKE_COMMAND}
                "-DEXECUTABLE=$<TARGET_FILE:oled_simulator>"
                -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_demo.cmake")
            set_tests_properties(demo_smoke PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
        endif()
    endif()
endif()

install(TARGETS oled_sim EXPORT oledTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT oledTargets NAMESPACE oled:: DESTINATION lib/cmake/oled_sim)
